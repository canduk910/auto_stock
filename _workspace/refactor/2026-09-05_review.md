  class KstDailyEmitCap(DailyEmitCap[K]):
      """KST 날짜가 바뀌면 첫 should_emit 에서 스스로 reset_daily. now 주입은 테스트 seam."""
      def __init__(self) -> None:
          super().__init__(); self._day = ""
      def should_emit(self, key: K, *, now: datetime | None = None) -> bool:
          today = (now or datetime.now(KST)).date().isoformat()
          if self._day != today:
              self._day = today; self.reset_daily()
          return super().should_emit(key)
      def emit_once(self, key: K, log_fn, msg: str, *args, now=None) -> bool:
          """peek → log_fn(msg, *args) → mark. log_fn 이 던지면 mark 안 함(D-3), 예외는 흡수하고 False."""
          try:
              if not self.should_emit(key, now=now): return False
              log_fn(msg, *args); self.mark_emitted(key); return True
          except Exception:
              return False
